$('input').on('paste', (event) => {
    // Use setTimeout to ensure the paste event has completed
    setTimeout(() => {
        let input = event.target;
        let value = input.value;
        // Replace all non-alphanumeric characters and spaces with an empty string
        let updatedValue = value.replace(/[^a-zA-Z0-9À-Ÿ\-,&()\'\’ ]/g, '');
        input.value = updatedValue;
    }, 0);
});
$('input').on('keypress', function (event) {
    var regex = new RegExp("^[a-zA-Z0-9À-Ÿ\-,&()\'\’ ]+$");
    var key = String.fromCharCode(!event.charCode ? event.which : event.charCode);
    if (!regex.test(key)) {
        event.preventDefault();
        return false;
    }
});
function BuildtheDatatable() {

    var selectCourt = $("#" + tabId + "selectCourt").val();
    var selectPartyName = $("#selectPartyName");
    var selectFrom = $("#selectFromDate");
    var selectTo = $("#selectToDate");
    var selectCourtNumber = $("#selectCourtNumber");
    var selectShip = $("#selectShip");
    var selectRelCaseCourtNo = $("#selectRetcaseCourtNumber");
    var selectIntPropName = $("#selectIntPropName");
    var selectIntPropNumber = $("#selectIntPropNumber");

    if (selectPartyName.val().length > 0) {
        name = selectPartyName.val();
        whatwasSearched = name;
        whatfieldwasSearched = (lang === "en" ? "Party Name: " : "Nom de la partie: ");
    }
    if (selectFrom.val().length > 0) {
        from = selectFrom.val();
        console.log(from);
    }
    if (selectTo.val().length > 0) {
        to = selectTo.val();
        console.log(to);
    }
    if (selectCourtNumber.val().length > 0) {
        courtnumber = selectCourtNumber.val();
        whatwasSearched = courtnumber;
        whatfieldwasSearched = (lang === "en" ? "Court Number: " : "Numéro de dossier: ");
    }
    if (selectShip.val().length > 0) {
        ship = selectShip.val();
        whatwasSearched = ship;
        whatfieldwasSearched = (lang === "en" ? "Ship: " : "Nom du navire: ");
    }
    if (selectIntPropName.val().length > 0 || selectIntPropNumber.val().length > 0) {
        name = selectIntPropName.val();
        number = selectIntPropNumber.val();
        whatwasSearched = name;
        whatfieldwasSearched = (lang === "en" ? "Intellectual Property: " : "Propriété intellectuelle: ");
    }
    if (selectRelCaseCourtNo.val().length > 0) {
        number = selectRelCaseCourtNo.val();
        whatwasSearched = number;
        whatfieldwasSearched = (lang === "en" ? "Related Cases: " : "Causes connexes: ");
    }

    court = selectCourt;
    selcourt = selectCourt;

    CourtNo = selectRelCaseCourtNo.val();

    //Which API to call
    switch (tabId) {
        case "tab01":
            whattab = "tab01";

            if (selectPartyName.val().length > 1) {
                $('#loadingscreen').show();

                data.ProceedingsQueriesPartyName(court, name.trim().replace(/\./g, 'xxyxx'), from, to, function (data) {
                    searchCallback(data);
                    $('#loadingscreen').hide();  // hide the loading message.
                    selectPartyName = '';
                    from = '';
                    to = '';
                });
            } else {
                if (selectShip.val().length > 0) {
                    $('#loadingscreen').show();
                    data.ProceedingsQueriesShip(court, ship.trim().replace(/\./g, 'xxyxx'), from, to, function (data) {
                        searchCallback(data);
                        $('#loadingscreen').hide();  // hide the loading message.
                        selectShip = '';
                        from = '';
                        to = '';
                    });
                }
            }
            break;

        case "tab02":
            whattab = "tab02";
            if (selectCourtNumber.val().length > 0) {
                $('#loadingscreen').show();
                data.ProceedingsQueriesCourtNumber(court, courtnumber.trim(), function (data) {
                    searchCallback(data);
                    $('#loadingscreen').hide();  // hide the loading message.
                    selectCourtNumber = '';
                });
            }
            break;

        case "tab03":
            whattab = "tab03";
            if (selectIntPropName.val().length > 0 || selectIntPropNumber.val().length > 0) {
                $('#loadingscreen').show();
                data.ProceedingsIntlProp(court, name.replace(/\./g, 'xxyxx'), number, function (data) {
                    searchCallback(data);
                    $('#loadingscreen').hide();  // hide the loading message.
                    selectIntPropName = '';
                });
            }
            break;

        case "tab04":
            whattab = "tab04";
            if (selectRelCaseCourtNo.val().length > 0) {
                $('#loadingscreen').show();
                data.ProceedingsRelCases(CourtNo, function (data) {
                    searchCallback(data);
                    $('#loadingscreen').hide();  //hide the loading message
                    selectRelCaseCourtNo = '';
                });
            }
            break;
    }
}

function SetupTabs() {

    tabId = "tab01";

    var $tabButtonItem = $('#tab-button li'),
        $tabSelect = $('#tab-select'),
        $tabContents = $('.tab-contents'),
        activeClass = 'is-active';

    $tabButtonItem.first().addClass(activeClass);
    $tabContents.not(':first').hide();

    $tabButtonItem.find('a').on('click', function (e) {
        var target = $(this).attr('href');

        $tabButtonItem.removeClass(activeClass);
        $(this).parent().addClass(activeClass);
        $tabSelect.val(target);
        $tabContents.hide();
        $(target).show();
        e.preventDefault();
    });

    $tabSelect.on('change', function () {
        var target = $(this).val(),
            targetSelectNum = $(this).prop('selectedIndex');

        $tabButtonItem.removeClass(activeClass);
        $tabButtonItem.eq(targetSelectNum).addClass(activeClass);
        $tabContents.hide();
        $(target).show();
    });

}

function SearchSummary() {

    var table = $('#datatable').DataTable();
    var info = table.page.info();


    (lang === "en" ? moment.locale('en') : moment.locale('fr'));

    var summary = "";
    var title = (lang === "en" ? "Search Summary" : "Résumé de recherche") + "\n";
    var date = moment().format('MMMM Do YYYY, h:mm:ss a');
    var court = WhatCourt();
    var what = whatfieldwasSearched + whatwasSearched;

    var c = info.recordsDisplay + (lang === 'en' ? " entries" : " éléments");

    summary = title + "\n" + date + "\n" + court + "\n" + what + "\n" + c;

    return summary;
}
function WhatCourt() {

    var c = "";
    switch ($("#" + tabId + "selectCourt").val()) {
        case "t":
            c = (lang === "en" ? "Federal Court" : "Cour fédérale");
            break;
        case "a":
            c = (lang === "en" ? "Federal Court of Appeal" : "Cour d’appel fédérale");
            break;
        case "b":
            c = (lang === "en" ? "Federal Court and Federal Court of Appeal" : "Cour fédérale et Cour d'appel fédérale");
            break;
        default:
    }
    return c;
}
